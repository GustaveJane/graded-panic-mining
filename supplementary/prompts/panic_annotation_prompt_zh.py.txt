def get_prompt_template_for_batch(template_type, texts):
    """
    批量Prompt模板（3种版本，均详细优化，贴合恐慌情绪标注）
    template_type: keyword/fewshot/example
    texts: 批量文本列表（一次10条）
    返回：符合OpenAI调用格式的prompt
    """
    # 批量文本格式化（统一格式，便于模型识别）
    items = "\n".join([f"{i + 1}. 文本：{str(txt)[:CONFIG['MAX_TEXT_LENGTH']]}" for i, txt in enumerate(texts)])
    # 基础系统提示（所有版本共用，明确任务）
    base_system = "你是一名专业的新冠疫情社交媒体文本情绪分析员，任务是精准判断每条文本是否包含【由新冠疫情直接或间接引发的恐慌情绪】，仅做二分类判定，严格遵循输出要求。"

    # 版本1：keyword版（关键词导向，精准匹配，适合快速判定）
    if template_type == "keyword":
        prompt = f"""
{base_system}

2.  文本包含恐慌情绪相关关键词/特征（必须包含以下任意1个及以上，体现恐慌心理或行为）：
    （1）情绪关键词：恐惧、害怕、恐慌、焦虑、紧张、不安、惶恐、绝望、无助、崩溃、心慌、失眠、胸闷、发抖、失控
    （2）认知偏差关键词：完了、完蛋了、全完了、治不好、必死、没救了、世界末日、要疯了、熬不住了
    （3）恐慌行为关键词：抢药、囤药、疯抢、抢购、逃离、跑、躲、不敢出门、过度求医、连夜排队
    （4）生理反应关键词：心慌、胸闷、失眠、发抖、出汗、呼吸困难、食欲不振、精神恍惚

【排除标准（满足任意1条，均判定为无恐慌情绪，输出0）】
1.  仅客观陈述疫情相关事实、新闻报道、数据通报（无任何主观情绪），例如：“今日新增确诊10例”
2.  仅表达不满、愤怒、抱怨、悲伤等其他负面情绪（无恐慌特征），例如：“封控太影响生活，太生气了”
3.  文本主题与新冠疫情无关（即使有负面情绪，也判定为0）
4.  仅为理性求助、建议、科普（无恐慌情绪），例如：“请问哪里可以做核酸？”
5.  仅表达对疫情的希望、祝福（无恐慌情绪），例如：“希望疫情早点结束，大家都平安”
6.  无实质内容的口号、蹭流量文本（无恐慌情绪）

【输出要求（必须严格遵守，否则判定为无效结果）】
1.  仅输出一个严格的JSON数组，数组长度与输入文本条数完全一致，顺序一一对应
2.  数组内仅包含0和1两个数字，1=有新冠疫情引发的恐慌情绪，0=无新冠疫情引发的恐慌情绪
3.  无任何多余文字、解释、标点、空格，不添加示例、说明等内容
4.  示例：输入3条文本，输出格式为 [1,0,1]

{items}
